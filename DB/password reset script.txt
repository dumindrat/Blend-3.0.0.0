alter table auth_user add PasswordUpdateDate date
go
create procedure [dbo].[auth_UserUpdatePassword] (@UserNme varchar(45),@Password varchar(45))
as

update auth_User set Password = @Password,PasswordExpireDate= dateadd(day,21,PasswordUpdateDate)
where UserName = @UserNme

update auth_User set PasswordUpdateDate= getdate()
where UserName = @UserNme
go

ALTER   PROCEDURE  [dbo].[auth_UserGet] (@LoginId varchar(45))
as
BEGIN
SELECT    Id ,
       UserName ,
       Password ,
       Email ,
       Name ,
       IsActive ,
       CreateDate ,
       UpdateDate,
       convert(varchar,PasswordExpireDate,101)PasswordExpireDate
FROM  auth_user  where UserName=@LoginId;

END
go
